step 1 --> install AutoMapper from nugetpackagemanger for all project or for service and APi layer
step 2 --> Create Class such as 
  -->  MappingProfile  (it is used while registering mapper with our APi application in Program.cs file )

using AutoMapper;
using TDM.Data.Entities;
using TDM.Service.DTOs;

public class MappingProfile : Profile
{
    public MappingProfile()
    {

        CreateMap<Role, RoleDto>();
        CreateMap<RoleDto, Role>();

        CreateMap<UserTask, TaskDto>();
        CreateMap<TaskDto, UserTask>();
    }
}

step 3 --->
	in program.cs file 
	do following after builder creation 
            var builder = WebApplication.CreateBuilder(args);

            // Register AutoMapper
            builder.Services.AddAutoMapper(typeof(MappingProfile)); // Register AutoMapper using Program type

            // Register your services  // add services here where you want to use mapper 
            builder.Services.AddTransient<RoleManager>();    // Register RoleService
            builder.Services.AddTransient<TaskManager>(); // Register UserTaskService
